// ==== unity requirements ====
Unity側で必要な設定
- 位置・回転・物理力をスクリプトから変更するため、このアイテムは移動可能/物理操作可能な設定が必要です。
// ==== code ====
// ── 毎フレーム ──
$.onUpdate((deltaTime) => {
  // 「位置を x:0 y:2 z:0 だけ 1 秒かけて動かす」の途中の動き
  if ($.state["__jpp_smooth_move_b2_active"]) {
    const read = (key, fallback) => (typeof $.state[key] === "number" ? $.state[key] : fallback);
    const delta = typeof deltaTime === "number" ? Math.max(0, deltaTime) : 0; // このフレームの経過時間
    const duration = Math.max(0.001, read("__jpp_smooth_move_b2_duration", 0.001));
    const current = $.getPosition() || new Vector3(0, 0, 0);
    const fromX = read("__jpp_smooth_move_b2_from_x", current.x);
    const fromY = read("__jpp_smooth_move_b2_from_y", current.y);
    const fromZ = read("__jpp_smooth_move_b2_from_z", current.z);
    const toX = read("__jpp_smooth_move_b2_to_x", fromX);
    const toY = read("__jpp_smooth_move_b2_to_y", fromY);
    const toZ = read("__jpp_smooth_move_b2_to_z", fromZ);
    const elapsed = Math.min(duration, read("__jpp_smooth_move_b2_elapsed", 0) + delta);
    const t = elapsed / duration; // 0（始点）〜 1（終点）
    $.setPosition(new Vector3(
      fromX + (toX - fromX) * t,
      fromY + (toY - fromY) * t,
      fromZ + (toZ - fromZ) * t
    ));
    $.state["__jpp_smooth_move_b2_elapsed"] = elapsed;
    if (elapsed >= duration) {
      $.state["__jpp_smooth_move_b2_active"] = false;
      $.setPosition(new Vector3(toX, toY, toZ));
    }
  }

  // 「角度を x:0 y:90 z:0 だけ 2 秒かけて回す」の途中の動き
  if ($.state["__jpp_smooth_rotq_b7_active"]) {
    const read = (key, fallback) => (typeof $.state[key] === "number" ? $.state[key] : fallback);
    const delta = typeof deltaTime === "number" ? Math.max(0, deltaTime) : 0; // このフレームの経過時間
    const duration = Math.max(0.001, read("__jpp_smooth_rotq_b7_duration", 0.001));
    const from = new Quaternion(read("__jpp_smooth_rotq_b7_from_qx", 0), read("__jpp_smooth_rotq_b7_from_qy", 0), read("__jpp_smooth_rotq_b7_from_qz", 0), read("__jpp_smooth_rotq_b7_from_qw", 1));
    const elapsed = Math.min(duration, read("__jpp_smooth_rotq_b7_elapsed", 0) + delta);
    const t = elapsed / duration; // 0（始め）〜 1（終わり）
    // 「角度を〜ずつ変える」と同じ回し方で、回す量を t の割合だけ進める
    const step = new Vector3(read("__jpp_smooth_rotq_b7_delta_x", 0) * t, read("__jpp_smooth_rotq_b7_delta_y", 0) * t, read("__jpp_smooth_rotq_b7_delta_z", 0) * t);
    $.setRotation(from.multiply(new Quaternion().setFromEulerAngles(step)));
    $.state["__jpp_smooth_rotq_b7_elapsed"] = elapsed;
    if (elapsed >= duration) {
      $.state["__jpp_smooth_rotq_b7_active"] = false;
    }
  }
});

// ── インタラクト時 ──
$.onInteract((player) => {
  // 位置を x:0 y:2 z:0 だけ 1 秒かけて動かす
  {
    // 動き始め: 始点と終点を覚えて、毎フレームの処理で少しずつ動かす
    const from = $.getPosition() || new Vector3(0, 0, 0);
    const to = from.clone().add(new Vector3(0, 2, 0));
    $.state["__jpp_smooth_move_b2_active"] = true;
    $.state["__jpp_smooth_move_b2_elapsed"] = 0;
    $.state["__jpp_smooth_move_b2_duration"] = 1;
    $.state["__jpp_smooth_move_b2_from_x"] = from.x;
    $.state["__jpp_smooth_move_b2_from_y"] = from.y;
    $.state["__jpp_smooth_move_b2_from_z"] = from.z;
    $.state["__jpp_smooth_move_b2_to_x"] = to.x;
    $.state["__jpp_smooth_move_b2_to_y"] = to.y;
    $.state["__jpp_smooth_move_b2_to_z"] = to.z;
  }
  // 角度を x:0 y:90 z:0 だけ 2 秒かけて回す
  {
    // 回り始め: 今の向きと回す量を覚えて、毎フレームの処理で少しずつ回す
    const from = $.getRotation() || new Quaternion();
    $.state["__jpp_smooth_rotq_b7_active"] = true;
    $.state["__jpp_smooth_rotq_b7_elapsed"] = 0;
    $.state["__jpp_smooth_rotq_b7_duration"] = 2;
    $.state["__jpp_smooth_rotq_b7_from_qx"] = from.x;
    $.state["__jpp_smooth_rotq_b7_from_qy"] = from.y;
    $.state["__jpp_smooth_rotq_b7_from_qz"] = from.z;
    $.state["__jpp_smooth_rotq_b7_from_qw"] = from.w;
    $.state["__jpp_smooth_rotq_b7_delta_x"] = 0;
    $.state["__jpp_smooth_rotq_b7_delta_y"] = 90;
    $.state["__jpp_smooth_rotq_b7_delta_z"] = 0;
  }
});
