// ==== code ====
$.onUpdate((deltaTime) => {
  // Jcreate++ smooth movement
  {
    if ($.state["__jpp_smooth_move_b2_active"]) {
      const delta = (typeof deltaTime === "number") ? Math.max(0, deltaTime) : 0;
      const duration = Math.max(0.001, (typeof $.state["__jpp_smooth_move_b2_duration"] === "number" ? $.state["__jpp_smooth_move_b2_duration"] : 0.001));
      const current = $.getPosition() || new Vector3(0, 0, 0);
      const fromX = (typeof $.state["__jpp_smooth_move_b2_from_x"] === "number") ? $.state["__jpp_smooth_move_b2_from_x"] : current.x;
      const fromY = (typeof $.state["__jpp_smooth_move_b2_from_y"] === "number") ? $.state["__jpp_smooth_move_b2_from_y"] : current.y;
      const fromZ = (typeof $.state["__jpp_smooth_move_b2_from_z"] === "number") ? $.state["__jpp_smooth_move_b2_from_z"] : current.z;
      const toX = (typeof $.state["__jpp_smooth_move_b2_to_x"] === "number") ? $.state["__jpp_smooth_move_b2_to_x"] : fromX;
      const toY = (typeof $.state["__jpp_smooth_move_b2_to_y"] === "number") ? $.state["__jpp_smooth_move_b2_to_y"] : fromY;
      const toZ = (typeof $.state["__jpp_smooth_move_b2_to_z"] === "number") ? $.state["__jpp_smooth_move_b2_to_z"] : fromZ;
      const elapsed = Math.min(duration, (typeof $.state["__jpp_smooth_move_b2_elapsed"] === "number" ? $.state["__jpp_smooth_move_b2_elapsed"] : 0) + delta);
      const t = elapsed / duration;
      $.setPosition(new Vector3(
        fromX + ((toX - fromX) * t),
        fromY + ((toY - fromY) * t),
        fromZ + ((toZ - fromZ) * t)
      ));
      $.state["__jpp_smooth_move_b2_elapsed"] = elapsed;
      if (elapsed >= duration) {
        $.state["__jpp_smooth_move_b2_active"] = false;
        $.setPosition(new Vector3(toX, toY, toZ));
      }
    }
  }
  // Jcreate++ smooth rotation
  {
    if ($.state["__jpp_smooth_rotq_b7_active"]) {
      const delta = (typeof deltaTime === "number") ? Math.max(0, deltaTime) : 0;
      const duration = Math.max(0.001, (typeof $.state["__jpp_smooth_rotq_b7_duration"] === "number" ? $.state["__jpp_smooth_rotq_b7_duration"] : 0.001));
      const num = (key) => (typeof $.state[key] === "number") ? $.state[key] : 0;
      const from = new Quaternion(num("__jpp_smooth_rotq_b7_from_qx"), num("__jpp_smooth_rotq_b7_from_qy"), num("__jpp_smooth_rotq_b7_from_qz"), (typeof $.state["__jpp_smooth_rotq_b7_from_qw"] === "number") ? $.state["__jpp_smooth_rotq_b7_from_qw"] : 1);
      const elapsed = Math.min(duration, num("__jpp_smooth_rotq_b7_elapsed") + delta);
      const t = elapsed / duration;
      // Same as "rotate_by": start rotation * euler(delta), with the delta scaled by progress.
      $.setRotation(from.multiply(new Quaternion().setFromEulerAngles(new Vector3(num("__jpp_smooth_rotq_b7_delta_x") * t, num("__jpp_smooth_rotq_b7_delta_y") * t, num("__jpp_smooth_rotq_b7_delta_z") * t))));
      $.state["__jpp_smooth_rotq_b7_elapsed"] = elapsed;
      if (elapsed >= duration) {
        $.state["__jpp_smooth_rotq_b7_active"] = false;
      }
    }
  }
});

$.onInteract((player) => {
  {
    const from = $.getPosition() || new Vector3(0, 0, 0);
    const to = from.clone().add(new Vector3(0, 2, 0));
    $.state["__jpp_smooth_move_b2_active"] = true;
    $.state["__jpp_smooth_move_b2_elapsed"] = 0;
    $.state["__jpp_smooth_move_b2_duration"] = Math.max(0.001, (1));
    $.state["__jpp_smooth_move_b2_from_x"] = from.x;
    $.state["__jpp_smooth_move_b2_from_y"] = from.y;
    $.state["__jpp_smooth_move_b2_from_z"] = from.z;
    $.state["__jpp_smooth_move_b2_to_x"] = to.x;
    $.state["__jpp_smooth_move_b2_to_y"] = to.y;
    $.state["__jpp_smooth_move_b2_to_z"] = to.z;
  }
  {
    const from = $.getRotation() || new Quaternion();
    $.state["__jpp_smooth_rotq_b7_active"] = true;
    $.state["__jpp_smooth_rotq_b7_elapsed"] = 0;
    $.state["__jpp_smooth_rotq_b7_duration"] = Math.max(0.001, (2));
    $.state["__jpp_smooth_rotq_b7_from_qx"] = from.x;
    $.state["__jpp_smooth_rotq_b7_from_qy"] = from.y;
    $.state["__jpp_smooth_rotq_b7_from_qz"] = from.z;
    $.state["__jpp_smooth_rotq_b7_from_qw"] = from.w;
    $.state["__jpp_smooth_rotq_b7_delta_x"] = (0);
    $.state["__jpp_smooth_rotq_b7_delta_y"] = (90);
    $.state["__jpp_smooth_rotq_b7_delta_z"] = (0);
  }
});
