// ==== unity requirements ====
Unity側で必要な設定
- アイテムに Movable Item を付けてください（位置・回転・物理の力をスクリプトから変えるため）。
- インタラクト（クリック・使う）するには、アイテムにColliderが必要です。
// ==== code ====
// ── 毎フレーム ──
$.onUpdate((deltaTime) => {
  deltaTime = typeof deltaTime === "number" && isFinite(deltaTime) ? Math.max(0, deltaTime) : 0; // このフレームの経過時間（秒）。おかしな値は 0 として扱う

  // 「位置を x:0 y:2 z:0 だけ 1 秒かけて動かす」の途中の動き
  if ($.state["__jpp_smooth_pos_b2_active"]) {
    const duration = $.state["__jpp_smooth_pos_b2_duration"];
    const elapsed = Math.min(duration, $.state["__jpp_smooth_pos_b2_elapsed"] + deltaTime);
    $.state["__jpp_smooth_pos_b2_elapsed"] = elapsed;
    const from = $.state["__jpp_smooth_pos_b2_from"];
    const to = $.state["__jpp_smooth_pos_b2_to"];
    // 始点から終点へ、経った時間の割合（0〜1）だけ進める。最後はちょうど終点に置く
    $.setPosition(elapsed < duration ? from.clone().lerp(to, elapsed / duration) : to);
    if (elapsed >= duration) {
      $.state["__jpp_smooth_pos_b2_active"] = false;
    }
  }

  // 「角度を x:0 y:90 z:0 だけ 2 秒かけて回す」の途中の動き
  if ($.state["__jpp_smooth_rot_b7_active"]) {
    const duration = $.state["__jpp_smooth_rot_b7_duration"];
    const elapsed = Math.min(duration, $.state["__jpp_smooth_rot_b7_elapsed"] + deltaTime);
    $.state["__jpp_smooth_rot_b7_elapsed"] = elapsed;
    const delta = $.state["__jpp_smooth_rot_b7_delta"];
    const from = $.state["__jpp_smooth_rot_b7_from"];
    // 「角度を〜だけ変える」と同じ回し方で、回す量を 経った時間の割合（0〜1）だけ進める
    const turned = new Quaternion().setFromEulerAngles(delta.clone().multiplyScalar(elapsed / duration));
    $.setRotation(from.clone().multiply(turned));
    if (elapsed >= duration) {
      $.state["__jpp_smooth_rot_b7_active"] = false;
    }
  }
});

// ── インタラクト時 ──
$.onInteract((player) => {
  // 位置を x:0 y:2 z:0 だけ 1 秒かけて動かす
  {
    // 動き始め: 始点と終点を覚えて、毎フレーム少しずつ動かす
    const from = $.getPosition() || new Vector3(0, 0, 0);
    $.state["__jpp_smooth_pos_b2_active"] = true;
    $.state["__jpp_smooth_pos_b2_elapsed"] = 0;
    $.state["__jpp_smooth_pos_b2_duration"] = 1;
    $.state["__jpp_smooth_pos_b2_from"] = from;
    $.state["__jpp_smooth_pos_b2_to"] = from.clone().add(new Vector3(0, 2, 0));
  }
  // 角度を x:0 y:90 z:0 だけ 2 秒かけて回す
  {
    // 回り始め: 今の向きと回す量を覚えて、毎フレーム少しずつ回す
    $.state["__jpp_smooth_rot_b7_active"] = true;
    $.state["__jpp_smooth_rot_b7_elapsed"] = 0;
    $.state["__jpp_smooth_rot_b7_duration"] = 2;
    $.state["__jpp_smooth_rot_b7_from"] = $.getRotation() || new Quaternion();
    $.state["__jpp_smooth_rot_b7_delta"] = new Vector3(0, 90, 0);
  }
});
