// ==== warnings ====
// ==== unity requirements ====
Unity側で必要な設定
- 位置・回転・物理力をスクリプトから変更するため、このアイテムは移動可能/物理操作可能な設定が必要です。
- インタラクト（クリック・使う）するには、アイテムにColliderが必要です。
// ==== code ====
// 位置を動かすブロックが複数あるので、このスクリプトが最後に決めた位置を覚えておき、そこに足していく
// （$.getPosition() は移動中の位置を返すので、同じフレームで何度も読むと、前の動きが消えてしまう）
// ただし今の位置が「最後に動かす前の位置 → 決めた位置」の線から 5cm 以上外れていたら、手で持つ・物理などで動いたので、今の位置から始める
const itemPosition = () => {
  const now = $.getPosition() || new Vector3(0, 0, 0);
  const p = $.state["__jpp_pose_position"];
  const from = $.state["__jpp_pose_position_from"];
  if (!p || !from) return now;
  // 線の上で、今の位置にいちばん近い点までの距離
  const line = p.clone().sub(from);
  const t = line.lengthSq() > 0 ? Math.min(1, Math.max(0, now.clone().sub(from).dot(line) / line.lengthSq())) : 1;
  if (now.clone().sub(from.clone().add(line.multiplyScalar(t))).length() > 0.05) return now;
  // 決めた位置に着いたら、線は終わり（このあとは決めた位置からのずれだけを見る）
  if (now.clone().sub(p).length() < 0.001) $.state["__jpp_pose_position_from"] = p;
  return p.clone();
};
// アイテムを pos に動かして、その位置と、動かす前の位置を覚える（すぐに着いたときは、動かす前の位置はいらない）
const placeItem = (pos) => {
  const before = $.getPosition() || new Vector3(0, 0, 0);
  $.setPosition(pos);
  const after = $.getPosition();
  $.state["__jpp_pose_position"] = pos;
  $.state["__jpp_pose_position_from"] = after && after.clone().sub(pos).length() < 0.001 ? pos : before;
};

// ── 開始時 ──
$.onStart(() => {
  // 角度を x:0 y:90 z:0 にする
  $.setRotation(new Quaternion().setFromEulerAngles(new Vector3(0, 90, 0)));
});

// ── 毎フレーム ──
$.onUpdate((deltaTime) => {
  deltaTime = typeof deltaTime === "number" && isFinite(deltaTime) ? Math.max(0, deltaTime) : 0; // このフレームの経過時間（秒）。おかしな値は 0 として扱う

  // 「向いている方向に 右:1 上:0 前:0 を 1 秒かけて進む」の途中の動き
  if ($.state["__jpp_smooth_pos_b11_active"]) {
    const duration = $.state["__jpp_smooth_pos_b11_duration"];
    const before = $.state["__jpp_smooth_pos_b11_elapsed"];
    const elapsed = Math.min(duration, before + deltaTime);
    $.state["__jpp_smooth_pos_b11_elapsed"] = elapsed;
    const from = $.state["__jpp_smooth_pos_b11_from"];
    const to = $.state["__jpp_smooth_pos_b11_to"];
    // 前のフレームから進んだ割合の分だけ動かす（ほかの動きと足し合わせるため）
    const move = to.clone().sub(from).multiplyScalar((elapsed - before) / duration);
    placeItem(itemPosition().add(move));
    if (elapsed >= duration) {
      $.state["__jpp_smooth_pos_b11_active"] = false;
    }
  }
});

// ── インタラクト時 ──
$.onInteract((player) => {
  // 向いている方向に 右:0 上:0 前:2 進む
  {
    const rot = $.getRotation() || new Quaternion();
    const pos = itemPosition();
    placeItem(pos.add(new Vector3(0, 0, 2).applyQuaternion(rot)));
  }
  // 向いている方向に 右:1 上:0 前:0 を 1 秒かけて進む
  {
    // 動き始め: 始点と終点を覚えて、毎フレーム少しずつ動かす
    const from = itemPosition();
    $.state["__jpp_smooth_pos_b11_active"] = true;
    $.state["__jpp_smooth_pos_b11_elapsed"] = 0;
    $.state["__jpp_smooth_pos_b11_duration"] = 1;
    $.state["__jpp_smooth_pos_b11_from"] = from;
    $.state["__jpp_smooth_pos_b11_to"] = from.clone().add(new Vector3(1, 0, 0).applyQuaternion($.getRotation() || new Quaternion()));
  }
});
