$.onStart(() => {
  $.state["var.score"] = 0;
  $.state["str.label"] = "hello";
  $.state["bool.ok"] = true;
  $.state["__jpp_flag_door"] = false;
});

$.onUpdate((deltaTime) => {
  {
    const __jpp_edge_cond_b35 = (($.state["__jpp_cd_tap"] || 0) <= 0);
    if (__jpp_edge_cond_b35 && !$.state["__jpp_edge_b35"]) {
      $.state["bool.ok"] = false;
      $.state["__jpp_edge_b35"] = true;
    }
    if (!__jpp_edge_cond_b35) {
      $.state["__jpp_edge_b35"] = false;
    }
  }
  // Jcreate++ cooldown timers
  $.state["__jpp_cd_tap"] = Math.max(0, ($.state["__jpp_cd_tap"] || 0) - deltaTime);
});

$.onInteract((player) => {
  if (!((($.state["__jpp_cd_tap"] || 0) > 0))) {
    $.state["var.score"] = ($.state["var.score"] || 0) + (Math.floor((Math.min((1), (6))) + Math.random() * ((Math.max((1), (6))) - (Math.min((1), (6))) + 1)));
    $.state["__jpp_cd_tap"] = Math.max(0, (2));
  }
  if (((($.state["var.score"] || 0) >= 10) && (!!$.state["bool.ok"] || !!$.state["__jpp_flag_door"]))) {
    $.state["__jpp_flag_door"] = !$.state["__jpp_flag_door"];
  } else {
    $.state["str.label"] = ($.state["str.label"] || "");
  }
});
