// ==== unity requirements ====
Unity側で必要な設定
- 乗り物ギミックを使うには、アイテムにRidable Itemを付け、Seat（座る位置）に子オブジェクトを設定してください。Colliderも必要です。
- 移動系テンプレートを使うため、このアイテムは移動可能な設定が必要です。
// ==== code ====
// 位置を動かすブロックが複数あるので、このスクリプトが最後に決めた位置を覚えておき、そこに足していく
// （$.getPosition() は移動中の位置を返すので、同じフレームで何度も読むと、前の動きが消えてしまう）
const itemPosition = () => { const p = $.state["__jpp_pose_position"]; return p ? p.clone() : ($.getPosition() || new Vector3(0, 0, 0)); };
// アイテムを pos に動かして、その位置を覚える
const placeItem = (pos) => { $.state["__jpp_pose_position"] = pos; $.setPosition(pos); };
// 向きを変えるブロックが複数あるので、このスクリプトが最後に決めた向きを覚えておき、そこから回していく
// （$.getRotation() は回転中の向きを返すので、同じフレームで何度も読むと、前の回転が消えてしまう）
const itemRotation = () => { const q = $.state["__jpp_pose_rotation"]; return q ? q.clone() : ($.getRotation() || new Quaternion()); };
// アイテムを rot の向きにして、その向きを覚える
const turnItem = (rot) => { $.state["__jpp_pose_rotation"] = rot; $.setRotation(rot); };

// ── 乗り物ギミック: 乗る・降りる・操作 ──
$.onRide((isGetOn, player) => {
  if (isGetOn) {
    $.state["__jpp_ride_active"] = true;
  } else {
    // 降りたら止める
    $.state["__jpp_ride_active"] = false;
    $.state["__jpp_ride_fwd"] = 0;
    $.state["__jpp_ride_upd"] = 0;
    $.state["__jpp_ride_trn"] = 0;
  }
});

// 前後・左右の入力（input.y: 前後、input.x: 左右）
$.onSteer((input, player) => {
  if ($.state["__jpp_ride_active"]) {
    $.state["__jpp_ride_fwd"] = input.y * 5;
    $.state["__jpp_ride_trn"] = input.x * 90;
  }
});

// 上下の入力
$.onSteerAdditionalAxis((input, player) => {
  if ($.state["__jpp_ride_active"]) {
    $.state["__jpp_ride_upd"] = input * 2;
  }
});

// ── 毎フレーム ──
$.onUpdate((deltaTime) => {
  deltaTime = typeof deltaTime === "number" && isFinite(deltaTime) ? Math.max(0, deltaTime) : 0; // このフレームの経過時間（秒）。おかしな値は 0 として扱う

  // 乗り物ギミック: 乗っている間、入力に合わせて動かす（進む向き: 従来の向き（アイテムの左 -X））
  if ($.state["__jpp_ride_active"]) {
    const pos = itemPosition();
    const rot = itemRotation();
    const turnAmount = ($.state["__jpp_ride_trn"] || 0) * deltaTime;
    const nextRot = rot.clone().multiply(new Quaternion().setFromEulerAngles(new Vector3(0, turnAmount, 0)));

    const direction = nextRot.clone().createEulerAngles();
    const forwardRadian = ((direction.y + 270) % 360) * Math.PI / 180;
    const forwardAmount = $.state["__jpp_ride_fwd"] || 0;
    const forwardVec = new Vector3(
      Math.sin(forwardRadian) * forwardAmount * deltaTime,
      0,
      Math.cos(forwardRadian) * forwardAmount * deltaTime
    );
    const upVec = new Vector3(0, 1, 0).multiplyScalar(($.state["__jpp_ride_upd"] || 0) * deltaTime);

    placeItem(pos.clone().add(forwardVec).add(upVec));
    turnItem(nextRot);
  }

  // 追いかけるギミック: いちばん近いプレイヤーの方を向いて近づく
  {
    const pos = itemPosition();
    let target = null;
    let nearest = Infinity;
    for (const player of $.getPlayersNear(pos, 10)) {
      const playerPos = player.getPosition();
      if (!playerPos) continue;
      const distance = playerPos.clone().sub(pos).length();
      if (distance < nearest) {
        nearest = distance;
        target = player;
      }
    }
    if (target && target.exists()) {
      const targetPos = target.getPosition();
      if (targetPos) {
        const dir = targetPos.clone().sub(pos);
        const length = dir.length();
        if (length > 0) {
          // 前（+Z）がプレイヤーを向くように回す
          const angle = Math.atan2(dir.x, dir.z) * 180 / Math.PI;
          turnItem(new Quaternion().setFromEulerAngles(new Vector3(0, angle, 0)));
          if (length > 1) {
            placeItem(pos.add(dir.normalize().multiplyScalar(3 * deltaTime)));
          }
        }
      }
    }
  }
});
